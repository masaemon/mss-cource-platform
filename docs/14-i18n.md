## ミドルウェア更新

```typescript
// middleware.ts
import createMiddleware from 'next-intl/middleware';
import { NextRequest, NextResponse } from 'next/server';
import { createServerClient } from '@supabase/ssr';

const intlMiddleware = createMiddleware({
  locales: ['ja', 'en'],
  defaultLocale: 'ja',
  localePrefix: 'as-needed', // デフォルトロケール（ja）はパスに含めない
});

export async function middleware(request: NextRequest) {
  // 多言語処理
  const response = intlMiddleware(request);

  // Supabase セッション更新
  let supabaseResponse = NextResponse.next({
    request,
  });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            request.cookies.set(name, value);
            supabaseResponse.cookies.set(name, value, options);
          });
        },
      },
    }
  );

  await supabase.auth.getUser();

  // 認証チェック...

  return supabaseResponse;
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
  ],
};
```

## 言語切り替えボタン

```typescript
// app/_components/locale-switcher.tsx
'use client';

import { useLocale } from 'next-intl';
import { useRouter, usePathname } from 'next/navigation';

export function LocaleSwitcher() {
  const locale = useLocale();
  const router = useRouter();
  const pathname = usePathname();

  const switchLocale = (newLocale: string) => {
    // パスからロケールを置換
    const newPath = pathname.replace(`/${locale}`, `/${newLocale}`);
    router.push(newPath);
  };

  return (
    <div className="flex gap-2">
      <button
        onClick={() => switchLocale('ja')}
        className={locale === 'ja' ? 'font-bold' : ''}
      >
        日本語
      </button>
      <button
        onClick={() => switchLocale('en')}
        className={locale === 'en' ? 'font-bold' : ''}
      >
        English
      </button>
    </div>
  );
}
```

## データベースフィールドの選択

```typescript
// app/_lib/utils/locale.ts
export function getLocalizedField<T extends Record<string, any>>(
  obj: T,
  fieldName: string,
  locale: string
): string {
  const localizedKey = `${fieldName}_${locale}` as keyof T;
  return (obj[localizedKey] as string) || obj[`${fieldName}_ja`] || '';
}

// 使用例
const course = await getCourse(id);
const title = getLocalizedField(course, 'title', locale);
const description = getLocalizedField(course, 'description', locale);
```

## 注意事項

- データベースの `_ja` / `_en` フィールドと連携
- デフォルトロケールは日本語（`ja`）
- URL構造: `/` (日本語), `/en` (英語)
- フォールバック: 英語がない場合は日本語を表示
